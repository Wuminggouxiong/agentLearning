这段代码本质上其实并不能算作是TreeOfThought，它并没有树结构，最多只能算作并行的CoT
我目前观察到的几个问题有：
- 重复尝试相同问题：propose每次都从`{problem}`从头生成，没有“已经想到第二步，下一步该干什么”的上下文
- 上一轮思考无法被后续引用：每条 messages 都是独立的 user 消息，没有任何节点保存"父路径"，评估阶段虽然把 paths 字符串塞进 prompt 了，但下一步生成又忘了它
- 串行调用 + 没有剪枝 + autoTry 还多烧一次 LLM 来决定数量 + 生成和评估都是整段自由文本，无法机器解析
````python
import os, json
from dataclasses import dataclass, field
from typing import Optional
from concurrent.futures import ThreadPoolExecutor
from openai import OpenAI

client = OpenAI(api_key=os.environ["DEEPSEEK_API_KEY"], base_url="https://api.deepseek.com")
MODEL = "deepseek-chat"   # 按你账号可用模型改

def chat(prompt: str, temperature: float = 0.7) -> str:
    return client.chat.completions.create(
        model=MODEL,
        messages=[{"role": "user", "content": prompt}],
        temperature=temperature,
    ).choices[0].message.content.strip()

# ---------- 1. 树节点：这就是你缺的"状态" ----------
@dataclass
class Thought:
    content: str
    parent: Optional["Thought"] = None
    score: float = 0.0
    critique: str = ""
    children: list["Thought"] = field(default_factory=list)

    def path(self) -> list["Thought"]:
        chain, node = [], self
        while node is not None:
            chain.append(node); node = node.parent
        return list(reversed(chain))

    def path_text(self) -> str:
        return "\n".join(f"  第{i+1}步：{t.content}" for i, t in enumerate(self.path()))

# ---------- 2. 扩展：基于"父路径"，而不是从头复述问题 ----------
def propose(problem: str, parent: Thought, k: int) -> list[str]:
    prompt = f"""你在用思维树分步解决下面的问题。

问题：{problem}
到目前为止的推理：
{parent.path_text() or "  （尚未开始）"}

请在已有推理的基础上，给出 {k} 个**不同的下一步思路**。
要求：
- 必须承接上文，不要重新从"如何规划学习"这类总述开始；
- 每条具体、可执行、彼此差异明显；
- 每条以 "###" 开头，第一行是标题，下面一两句说明。"""
    raw = chat(prompt)
    parts = [p.strip() for p in raw.split("###") if p.strip()]
    # 简单去重
    seen, out = set(), []
    for p in parts:
        key = p[:30]
        if key not in seen:
            seen.add(key); out.append(p)
    return out[:k] or [raw]

# ---------- 3. 评估：一次批量打分 + 结构化输出 ----------
def evaluate(problem: str, parent: Thought, candidates: list[str]) -> list[dict]:
    listing = "\n".join(f"{i+1}. {c}" for i, c in enumerate(candidates))
    prompt = f"""问题：{problem}
已有推理：
{parent.path_text()}

候选下一步：
{listing}

请对每条候选打分（0-10，越高表示越可行、越能推进目标、越不与上文重复）。
严格输出 JSON 数组，每项为 {{"index": 序号, "score": 分数, "critique": "一句话理由"}}。
只输出 JSON，不要解释。"""
    raw = chat(prompt, temperature=0)
    try:
        return json.loads(raw[raw.find("["): raw.rfind("]") + 1])
    except Exception:
        return [{"index": i+1, "score": 5.0, "critique": "解析失败，给默认分"}
                for i in range(len(candidates))]

# ---------- 4. 主循环：BFS + beam 剪枝 ----------
def tree_of_thought(problem: str, breadth: int = 3, depth: int = 3, beam: int = 2) -> dict:
    root = Thought(content=f"问题：{problem}")
    frontier, trace = [root], []

    for d in range(depth):
        # (a) 并行扩展当前层每个节点
        candidates = []
        with ThreadPoolExecutor(max_workers=max(1, len(frontier))) as ex:
            fut2node = {ex.submit(propose, problem, node, breadth): node for node in frontier}
            for fut, node in fut2node.items():
                for text in fut.result():
                    candidates.append(Thought(content=text, parent=node))

        # (b) 按父节点分组，批量评估（比逐个调省时）
        groups: dict[int, list[Thought]] = {}
        for t in candidates:
            groups.setdefault(id(t.parent), []).append(t)
        for g in groups.values():
            parent = g[0].parent
            for s in evaluate(problem, parent, [t.content for t in g]):
                i = s.get("index", 0) - 1
                if 0 <= i < len(g):
                    g[i].score = float(s.get("score", 0))
                    g[i].critique = s.get("critique", "")
                    parent.children.append(g[i])

        # (c) 全局剪枝：只留 top-beam
        candidates.sort(key=lambda t: t.score, reverse=True)
        frontier = candidates[:beam]
        trace.append([(t.content[:40], t.score) for t in frontier])
        if not frontier:
            break

    # (d) 用最优路径的完整上下文产出最终答案
    if not frontier:
        return {"answer": "未生成有效路径", "trace": trace}
    best = frontier[0]
    answer = chat(f"""问题：{problem}

经过评估的最优推理路径：
{best.path_text()}

请基于以上路径，给出最终完整、可执行的回答。""", temperature=0.3)
    return {"answer": answer, "best_path": [t.content for t in best.path()], "trace": trace}

if __name__ == "__main__":
    problem = """我需要在3个月内学习 Python，用于数据分析工作。
我每天只有2小时学习时间，有一定的编程基础（学过简单的 HTML/CSS）。应该如何规划学习路径？"""
    r = tree_of_thought(problem, breadth=3, depth=3, beam=2)
    for i, layer in enumerate(r["trace"], 1):
        print(f"--- 第 {i} 层保留 ---")
        for content, score in layer:
            print(f"  [{score:>4}] {content}")
    print("\n最优路径：")
    for s in r["best_path"]:
        print("  •", s[:80])
    print("\n最终回答：\n", r["answer"])
